cmake_minimum_required(VERSION 3.21)
project(cm5-home VERSION 0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick Network Multimedia)
qt_add_resources(APP_RESOURCES resources.qrc)
add_executable(cm5-home main.cpp jellyfinclient.cpp jellyfinclient.h ${APP_RESOURCES})
target_link_libraries(cm5-home PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::Network Qt6::Multimedia)
install(TARGETS cm5-home RUNTIME DESTINATION bin)
